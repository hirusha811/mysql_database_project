DELIMITER //

CREATE PROCEDURE coursebycourse(
    IN course_code_param VARCHAR(10)
)
BEGIN
    DECLARE done INT DEFAULT FALSE;
    DECLARE reg_no CHAR(10);
    DECLARE student_name VARCHAR(50);
    DECLARE total_sess INT;
    DECLARE present_cnt INT;
    DECLARE medical_count INT;
    DECLARE attendance_pct DECIMAL(5,2);
    DECLARE with_medical_pct DECIMAL(5,2);
    
    DECLARE student_cursor CURSOR FOR
    SELECT
        s.registration_no,
        s.full_name,
        COUNT(*) as total_sessions,
        SUM(CASE WHEN a.status = 'Present' THEN 1 ELSE 0 END) as present_count,
        SUM(CASE WHEN a.status = 'Medical' THEN 1 ELSE 0 END) as medical_count
    FROM attendance a
    JOIN student s ON a.registration_no = s.registration_no
    WHERE a.course = course_code_param
    GROUP BY s.registration_no, s.full_name;

    DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;

    CREATE TEMPORARY TABLE temp_attendance_categories (
        registration_no CHAR(10),
        student_name VARCHAR(50),
        total_sessions INT,
        present_count INT,
        medical_count INT,
        attendance_percentage_with_medical DECIMAL(5,2),
        attendance_percentage_without_medical DECIMAL(5,2),
        eligibility_with_medical VARCHAR(20),
        eligibility_without_medical VARCHAR(20)
    );

    OPEN student_cursor;

    student_loop: LOOP
        FETCH student_cursor INTO reg_no, student_name, total_sess, present_cnt, medical_count;

        IF done THEN
            LEAVE student_loop;
        END IF;

        SET with_medical_pct = ROUND(((present_cnt + medical_count) / total_sess) * 100, 2);
        SET attendance_pct = ROUND((present_cnt / total_sess) * 100, 2);

        INSERT INTO temp_attendance_categories VALUES (
            reg_no,
            student_name,
            total_sess,
            present_cnt,
            medical_count,
            with_medical_pct,                    
            attendance_pct,                    
            IF(with_medical_pct >= 80, 'ELIGIBLE', 'NOT ELIGIBLE'), 
            IF(attendance_pct >= 80, 'ELIGIBLE', 'NOT ELIGIBLE') 
        );

    END LOOP;

    CLOSE student_cursor;

    SELECT * FROM temp_attendance_categories
    ORDER BY registration_no;

    DROP TEMPORARY TABLE temp_attendance_categories;

END //

DELIMITER ;

CALL coursebycourse('ICT1233');
