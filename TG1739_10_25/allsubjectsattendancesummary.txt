DELIMITER //

CREATE PROCEDURE allsubjectsattendancesummary()
BEGIN
    SELECT 
        s.registration_no,
        s.full_name AS student_name,
        c.course_code,
        c.course_name,
        c.course_type,
        c.credits,
        l.full_name AS lecturer_name,
        COUNT(*) AS total_sessions,
		SUM(IF(a.status = 'Present', 1, 0)) as present_count,
        SUM(IF(a.status = 'Medical', 1, 0)) as medical_count,
        SUM(a.status = 'Absent') AS absent_count,
        ROUND((SUM(a.status IN ('Present', 'Medical')) / COUNT(*)) * 100, 2) AS attendance_percentage,
        CASE 
            WHEN ROUND((SUM(a.status IN ('Present', 'Medical')) / COUNT(*)) * 100, 2) >= 80 THEN 'ELIGIBLE'
            ELSE 'NOT ELIGIBLE'
        END AS eligibility
    FROM attendance a
    JOIN student s ON a.registration_no = s.registration_no
    JOIN course c ON a.course = c.course_code
    LEFT JOIN lecture l ON c.lecture = l.lecture_id
    GROUP BY s.registration_no, s.full_name, c.course_code, c.course_name, c.course_type, c.credits, l.full_name
    ORDER BY s.registration_no, c.course_code;
    
END //

DELIMITER ;


CALL allsubjectsattendancesummary();
