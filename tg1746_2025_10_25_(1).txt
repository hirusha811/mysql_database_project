--calculate SGPA and CGPA


CREATE OR REPLACE VIEW sgpa_cgpa AS
SELECT
    registration_no,
    ROUND(SUM(grade_points * credits) / SUM(credits), 2) AS sgpa, -- calculate part ofsgpa
    ROUND(SUM(grade_points * credits) / SUM(credits), 2) AS cgpa  -- calculate part of cgpa
FROM (
    SELECT
        totals.registration_no,
        totals.credits,
		
		
		-- 2 types for repeat and proper student grade_point calculate and suspended going to be 0
		
		
        ROUND(
            CASE
                WHEN totals.status = 'repeat' THEN
                    CASE
                        WHEN totals.total >= 80 THEN 2.0
                        WHEN totals.total >= 70 THEN 2.0
                        WHEN totals.total >= 65 THEN 2.0
                        WHEN totals.total >= 60 THEN 2.0
                        WHEN totals.total >= 55 THEN 2.0
                        WHEN totals.total >= 50 THEN 2.0
                        WHEN totals.total >= 45 THEN 2.0
                        WHEN totals.total >= 40 THEN 2.0
                        WHEN totals.total >= 35 THEN 1.7
                        WHEN totals.total >= 30 THEN 1.3
                        ELSE 0
                    END
                WHEN totals.status = 'proper' THEN
                    CASE
                        WHEN totals.total >= 85 THEN 4.0
                        WHEN totals.total >= 75 THEN 4.0
                        WHEN totals.total >= 70 THEN 3.7
                        WHEN totals.total >= 65 THEN 3.3
                        WHEN totals.total >= 60 THEN 3.0
                        WHEN totals.total >= 55 THEN 2.7
                        WHEN totals.total >= 50 THEN 2.3
                        WHEN totals.total >= 45 THEN 2.0
                        WHEN totals.total >= 40 THEN 1.7
                        WHEN totals.total >= 35 THEN 1.3
                        ELSE 0
                    END
		WHEN totals.status = 'suspended' THEN 0
            END, 2
        ) AS grade_points
    FROM (
        SELECT
            m.registration_no,
            c.credits,
            s.status,
			
			
	-- calculate the final marks
			
			
            CASE
                WHEN c.course_type = 'BOTH' THEN
                    ((COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END),0) +
                      COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END),0) +
                      COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END),0)
                     - LEAST(
                         COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END),0),
                         COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END),0),
                         COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END),0)
                     )) / 2) * 0.10
                    + ((COALESCE(MAX(CASE WHEN m.status='MT' THEN m.mark END),0) +
                        COALESCE(MAX(CASE WHEN m.status='MP' THEN m.mark END),0)) / 2) * 0.25
                    + ((COALESCE(MAX(CASE WHEN m.status='FT' THEN m.mark END),0) +
                        COALESCE(MAX(CASE WHEN m.status='FP' THEN m.mark END),0)) / 2) * 0.70
                    + (COALESCE(MAX(CASE WHEN m.status='A' THEN m.mark END),0) * 0.05)
                WHEN c.course_type = 'T' THEN
                    ((COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END),0) +
                      COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END),0) +
                      COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END),0)
                     - LEAST(
                         COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END),0),
                         COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END),0),
                         COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END),0)
                     )) / 2) * 0.10
                    + (COALESCE(MAX(CASE WHEN m.status='MT' THEN m.mark END),0) * 0.25)
                    + (COALESCE(MAX(CASE WHEN m.status='FT' THEN m.mark END),0) * 0.70)
                    + (COALESCE(MAX(CASE WHEN m.status='A' THEN m.mark END),0) * 0.05)
                WHEN c.course_type = 'P' THEN
                    ((COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END),0) +
                      COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END),0) +
                      COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END),0)
                     - LEAST(
                         COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END),0),
                         COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END),0),
                         COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END),0)
                     )) / 2) * 0.10
                    + (COALESCE(MAX(CASE WHEN m.status='MP' THEN m.mark END),0) * 0.25)
                    + (COALESCE(MAX(CASE WHEN m.status='FP' THEN m.mark END),0) * 0.70)
                    + (COALESCE(MAX(CASE WHEN m.status='A' THEN m.mark END),0) * 0.05)
                ELSE 0
            END AS total
        FROM marks m
        JOIN course c ON m.course_code = c.course_code
        JOIN student s ON m.registration_no = s.registration_no
        GROUP BY m.registration_no, c.course_name, c.credits, s.status, c.course_type
    ) AS totals
) AS grade_calculated
GROUP BY registration_no;
