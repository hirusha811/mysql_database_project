---user table---

CREATE TABLE user(
	user_id CHAR(10) PRIMARY KEY,
	user_name VARCHAR(50)NOT NULL,
	password VARCHAR(50) NOT NULL,
	role ENUM('lecture','dean','admin','technical_officer','student')NOT NULL,
	email VARCHAR(100),
	phone VARCHAR(15),
	department CHAR(10),
	appointment_start_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP
	
);

---updated student table---

CREATE TABLE student(
	registration_no CHAR(10) PRIMARY KEY,
	full_name VARCHAR(20) NOT NULL,
	user_id CHAR(10),
	assing_department CHAR(50),
    	dob DATE,
    	address VARCHAR(100),
    	batch_year INT,
    	status ENUM('proper','repeat','suspended') DEFAULT 'proper',
	FOREIGN KEY (user_id) REFERENCES user(user_id) ON DELETE CASCADE ON UPDATE CASCADE,
	FOREIGN KEY (assing_department) REFERENCES department(department_id) ON DELETE SET NULL ON UPDATE CASCADE
);

---updated medical table---

CREATE TABLE medical(
	   medical_id INT AUTO_INCREMENT PRIMARY KEY,
	   registration_no CHAR(10),
	   attendance_id INT,
	   leave_date DATE,
	   reason VARCHAR(50),
	   submitted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
	   approved_by CHAR(20) DEFAULT 'pending', 
	   FOREIGN KEY (registration_no) REFERENCES student(registration_no)ON DELETE CASCADE,
	   FOREIGN KEY (attendance_id) REFERENCES attendance(attendance_id) ON DELETE CASCADE
);
