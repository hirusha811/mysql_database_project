--course table

CREATE TABLE course(
	   course_code VARCHAR(10) PRIMARY KEY,
	   course_name VARCHAR(60) NOT NULL,
	   credits INT,
	   department CHAR(10),
	   course_type ENUM('T','P','BOTH'),   
	   hours INT,     
	   lecture CHAR(10),
	   FOREIGN KEY (lecture) REFERENCES lecture(lecture_id) ON DELETE SET NULL ON UPDATE CASCADE,
	   FOREIGN KEY (department) REFERENCES department(department_id)ON DELETE SET NULL ON UPDATE CASCADE
);

--marks table

CREATE TABLE marks(
	  marks_id INT PRIMARY KEY AUTO_INCREMENT,
	  registration_no CHAR(10),
	  course_code VARCHAR(10),
	  week_number INT,
	  status ENUM('Q1','Q2','Q3','MT','MP','FT','FP','A'),
	  mark INT CHECK (mark >= 0 AND mark <= 100),
	  recorded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
	  FOREIGN KEY (course_code) REFERENCES course(course_code)ON DELETE CASCADE,
	  FOREIGN KEY (registration_no) REFERENCES student(registration_no) ON DELETE CASCADE
);
