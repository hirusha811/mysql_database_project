CREATE TABLE lecture(
		lecture_id CHAR(10) PRIMARY KEY,
		user_id CHAR(10),
		full_name VARCHAR(40),
		qualifications VARCHAR(50),
		assing_department CHAR(50),
		office_room CHAR(30),
		FOREIGN KEY (user_id) REFERENCES user(user_id) ON DELETE CASCADE ON UPDATE CASCADE,
		FOREIGN KEY (assing_department) REFERENCES department(department_id) ON DELETE SET NULL ON UPDATE CASCADE
);

CREATE TABLE tech_officer(
		to_id CHAR(10) PRIMARY KEY,
		user_id CHAR(10),
		full_name VARCHAR(50),
		expert_role VARCHAR(60),
		assing_department CHAR(10),
		status ENUM('Active', 'Inactive', 'On Leave'),
		FOREIGN KEY (user_id) REFERENCES user(user_id) ON DELETE CASCADE ON UPDATE CASCADE,
		FOREIGN KEY (assing_department) REFERENCES department(department_id) ON DELETE SET NULL ON UPDATE CASCADE
);

CREATE TABLE attendance(
		attendance_id INT PRIMARY KEY AUTO_INCREMENT,
		course VARCHAR(10),
		registration_no CHAR(10) NOT NULL,
		hours INT,
		status ENUM('Absent','Present','Medical') DEFAULT 'Absent' NOT NULL,
		session_type ENUM('T','P') NOT NULL,
		week_number INT,
		attendance_mark_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
		recorded_by CHAR(10),
		FOREIGN KEY (course) REFERENCES course(course_code) ON DELETE CASCADE ON UPDATE CASCADE,
		FOREIGN KEY (registration_no) REFERENCES student(registration_no) ON DELETE CASCADE,
		FOREIGN KEY (recorded_by) REFERENCES tech_officer(to_id) ON DELETE SET NULL
);
