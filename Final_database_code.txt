CREATE DATABASE fotdb;
USE fotdb;


CREATE TABLE user(
	user_id CHAR(10) PRIMARY KEY,
	user_name VARCHAR(50)NOT NULL,
	password VARCHAR(50) NOT NULL,
	role ENUM('lecture','dean','admin','technical_officer','student')NOT NULL,
	email VARCHAR(100),
	phone VARCHAR(15),
	department CHAR(10),
	appointment_start_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP
	
);

CREATE TABLE department(
    	department_id CHAR(10) PRIMARY KEY,
		department_name VARCHAR(20),
		location VARCHAR(30),
		establish_date DATE,
		budget FLOAT(8,2)
);
 
CREATE TABLE student(
	registration_no CHAR(10) PRIMARY KEY,
	full_name VARCHAR(20) NOT NULL,
	user_id CHAR(10),
	assing_department CHAR(50),
    dob DATE,
    address VARCHAR(100),
    batch_year INT,
    status ENUM('proper','repeat','suspended') DEFAULT 'proper',
	FOREIGN KEY (user_id) REFERENCES user(user_id) ON DELETE CASCADE ON UPDATE CASCADE,
	FOREIGN KEY (assing_department) REFERENCES department(department_id) ON DELETE SET NULL ON UPDATE CASCADE
);



CREATE TABLE lecture(
		lecture_id CHAR(10) PRIMARY KEY,
		user_id CHAR(10),
		full_name VARCHAR(40),
		qualifications VARCHAR(50),
		assing_department CHAR(50),
		office_room CHAR(30),
		FOREIGN KEY (user_id) REFERENCES user(user_id) ON DELETE CASCADE ON UPDATE CASCADE,
		FOREIGN KEY (assing_department) REFERENCES department(department_id) ON DELETE SET NULL ON UPDATE CASCADE
);

CREATE TABLE tech_officer(
		to_id CHAR(10) PRIMARY KEY,
		user_id CHAR(10),
		full_name VARCHAR(50),
		expert_role VARCHAR(60),
		assing_department CHAR(10),
		status ENUM('Active', 'Inactive', 'On Leave'),
		FOREIGN KEY (user_id) REFERENCES user(user_id) ON DELETE CASCADE ON UPDATE CASCADE,
		FOREIGN KEY (assing_department) REFERENCES department(department_id) ON DELETE SET NULL ON UPDATE CASCADE
);

CREATE TABLE course(
	   course_code VARCHAR(10) PRIMARY KEY,
	   course_name VARCHAR(60) NOT NULL,
	   credits INT,
	   department CHAR(10),
	   course_type ENUM('T','P','BOTH'),   
	   hours INT,     
	   lecture CHAR(10),
	   FOREIGN KEY (lecture) REFERENCES lecture(lecture_id) ON DELETE SET NULL ON UPDATE CASCADE,
	   FOREIGN KEY (department) REFERENCES department(department_id)ON DELETE SET NULL ON UPDATE CASCADE
);

CREATE TABLE attendance(
		attendance_id INT PRIMARY KEY AUTO_INCREMENT,
		course VARCHAR(10),
		registration_no CHAR(10) NOT NULL,
		hours INT,
		status ENUM('Absent','Present','Medical') DEFAULT 'Absent' NOT NULL,
		session_type ENUM('T','P') NOT NULL,
		week_number INT,
		attendance_mark_time TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
		recorded_by CHAR(10),
		FOREIGN KEY (course) REFERENCES course(course_code) ON DELETE CASCADE ON UPDATE CASCADE,
		FOREIGN KEY (registration_no) REFERENCES student(registration_no) ON DELETE CASCADE,
		FOREIGN KEY (recorded_by) REFERENCES tech_officer(to_id) ON DELETE SET NULL
);

CREATE TABLE medical(
	   medical_id INT AUTO_INCREMENT PRIMARY KEY,
	   registration_no CHAR(10),
	   attendance_id INT,
	   leave_date DATE,
	   reason VARCHAR(50),
	   submitted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
	   approval_status CHAR(20) DEFAULT 'pending', 
	   FOREIGN KEY (registration_no) REFERENCES student(registration_no)ON DELETE CASCADE,
	   FOREIGN KEY (attendance_id) REFERENCES attendance(attendance_id) ON DELETE CASCADE
);

CREATE TABLE stu_course(
	   registration_no CHAR(10),
	   course CHAR(10),
	   PRIMARY KEY (registration_no, course),
	   FOREIGN KEY (registration_no) REFERENCES student(registration_no)ON DELETE CASCADE,
	   FOREIGN KEY (course) REFERENCES course(course_code) ON DELETE CASCADE
);

CREATE TABLE marks(
	  marks_id INT PRIMARY KEY AUTO_INCREMENT,
	  registration_no CHAR(10),
	  course_code VARCHAR(10),
	  week_number INT,
	  status ENUM('Q1','Q2','Q3','MT','MP','FT','FP','A'),
	  mark INT CHECK (mark >= 0 AND mark <= 100),
	  recorded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
	  FOREIGN KEY (course_code) REFERENCES course(course_code)ON DELETE CASCADE,
	  FOREIGN KEY (registration_no) REFERENCES student(registration_no) ON DELETE CASCADE
);

CREATE PROCEDURE ca_individual(IN reg_no CHAR(10))
BEGIN
    DECLARE total_courses INT DEFAULT 0;
    DECLARE count INT DEFAULT 0;
    DECLARE curren_course VARCHAR(10);

    DECLARE q1, q2, q3 DECIMAL(5,2);
    DECLARE mt, mp, midsum, assesment, qmax, qsum, final_ca DECIMAL(6,2);
    DECLARE ca_status VARCHAR(20);
	DECLARE sub_status VARCHAR(20);
	DECLARE stu_status CHAR(10);

    -- tempory table for result
    DROP TEMPORARY TABLE IF EXISTS temp_result;
    CREATE TEMPORARY TABLE temp_result (
        Registration_No CHAR(10),
        Course_Code VARCHAR(10),
        Q1 VARCHAR(5),
        Q2 VARCHAR(5),
        Q3 VARCHAR(5),
        Mid_Marks VARCHAR(10),
        Assignment_Marks VARCHAR(10)
    );


   -- get total count of courses
    SELECT COUNT(DISTINCT course_code) INTO total_courses
    FROM marks
    WHERE registration_no = reg_no;

SELECT status INTO stu_status
FROM student
WHERE registration_no = reg_no;

	
	-- If student is suspended, fill WH for all

	IF (stu_status='suspended') THEN	
		INSERT INTO temp_result
		SELECT 
			reg_no,
			course_code,
			'WH' AS Q1,
			'WH' AS Q2,
			'WH' AS Q3,
			'WH' AS Mid_Marks,
			'WH' AS Assesment_Marks
		FROM marks
		WHERE registration_no = reg_no
		GROUP BY course_code;

	ELSE
		
		WHILE count < total_courses DO
			
			SELECT DISTINCT course_code INTO curren_course
			FROM marks
			WHERE registration_no = reg_no
			LIMIT 1 OFFSET count;

			SELECT IFNULL(MAX(mark),0) * 0.05 INTO q1
			FROM marks
			WHERE registration_no = reg_no AND course_code = curren_course AND status='Q1';

			SELECT IFNULL(MAX(mark),0)*0.05 INTO q2
			FROM marks
			WHERE registration_no = reg_no AND course_code = curren_course AND status='Q2';

			SELECT IFNULL(MAX(mark),0)*0.05 INTO q3
			FROM marks
			WHERE registration_no = reg_no AND course_code = curren_course AND status='Q3';

		    SELECT IFNULL(MAX(mark),0) INTO mt
			FROM marks
			WHERE registration_no = reg_no AND course_code = curren_course AND status='MT';

			SELECT IFNULL(MAX(mark),0) INTO mp
			FROM marks
			WHERE registration_no = reg_no AND course_code = curren_course AND status='MP';

			SELECT IFNULL(MAX(mark),0)*0.05 INTO assesment
			FROM marks
			WHERE registration_no = reg_no AND course_code = curren_course AND status='A';
			
			SELECT course.course_type INTO sub_status
			FROM course
			WHERE course.course_code = curren_course; 
		

	
			-- set best 2 quizzes
			IF (q1 < q2 AND q1 < q3) THEN
				SET qmax = q2 + q3;
			ELSEIF (q2 < q3 AND q2 < q1) THEN
				SET qmax = q3 + q1;
			ELSE
				SET qmax = q1 + q2;
			END IF;
		

	
			SET qsum = (qmax/2);



			-- mid marks based on course type
			IF (sub_status='BOTH') THEN
				SET midsum = ((mt + mp)/2) * 0.20;
			ELSEIF (sub_status='T') THEN
				SET midsum = mt * 0.20;
			ELSEIF (sub_status='P') THEN
				SET midsum = mp * 0.20;
			END IF;


			-- final ca marks
			SET final_ca = qsum + midsum + assesment;


			-- pass/fail
			IF final_ca >= 14 THEN
				SET ca_status = 'CA_Pass';
			ELSE
				SET ca_status = 'CA_Fail';
			END IF;



			-- insert into temp_result
			INSERT INTO temp_result
			 VALUES (
				reg_no, 
				curren_course, 
				FORMAT(q1,2), 
				FORMAT(q2,2), 
				FORMAT(q3,2), 
				FORMAT(midsum,2), 
				FORMAT(assesment,2)
			 );

			SET count = count + 1;
		END WHILE;
	END IF;

    -- final output
    SELECT * FROM temp_result ORDER BY Course_Code;

    DROP TEMPORARY TABLE IF EXISTS temp_result;
END//

	SELECT COUNT(DISTINCT registration_no) INTO stucount
	FROM student;
	
WHILE maincount < stucount DO

	LIMIT 1 OFFSET maincount;
	
	SELECT status INTO stu_status

	SELECT COUNT(DISTINCT course_code) INTO total_course

	SET count = 0;

		FROM marks m
		JOIN course c ON m.course_code = c.course_code
		WHERE m.registration_no = reg_no
		GROUP BY m.course_code, c.course_name;

	ELSE
		
		WHILE count < total_course DO
		
			SELECT DISTINCT course_code INTO curren_course
			FROM marks
			WHERE registration_no = reg_no
			LIMIT 1 OFFSET count;
			
			SELECT course_name INTO cname
			FROM course
			WHERE course_code=curren_course;

			SELECT IFNULL(MAX(mark),0) INTO q1
			FROM marks
			WHERE registration_no = reg_no AND course_code = curren_course AND status='Q1';

			SELECT IFNULL(MAX(mark),0) INTO q2
			FROM marks
			WHERE registration_no = reg_no AND course_code = curren_course AND status='Q2';

			SELECT IFNULL(MAX(mark),0) INTO q3
			FROM marks
			WHERE registration_no = reg_no AND course_code = curren_course AND status='Q3';

		    SELECT IFNULL(MAX(mark),0) INTO mt
			FROM marks
			WHERE registration_no = reg_no AND course_code = curren_course AND status='MT';

			SELECT IFNULL(MAX(mark),0) INTO mp
			FROM marks
			WHERE registration_no = reg_no AND course_code = curren_course AND status='MP';

			SELECT IFNULL(MAX(mark),0)*0.05 INTO assesment
			FROM marks
			WHERE registration_no = reg_no AND course_code = curren_course AND status='A';
			
			SELECT course.course_type INTO sub_status
			FROM course
			WHERE course.course_code = curren_course; 
		
	

			-- set best 2 quizzes
			IF (q1 < q2 AND q1 < q3) THEN
				SET qmax = q2 + q3;
			ELSEIF (q2 < q3 AND q2 < q1) THEN
				SET qmax = q3 + q1;
			ELSE
				SET qmax = q1 + q2;
			END IF;

			SET qsum = (qmax/2) * 0.10;



			-- mid marks based on course type
			IF (sub_status='BOTH') THEN
				SET midsum = ((mt + mp)/2) * 0.20;
			ELSEIF (sub_status='T') THEN
				SET midsum = mt * 0.20;
			ELSEIF (sub_status='P') THEN
				SET midsum = mp * 0.20;
			END IF;



			-- final ca marks
			SET final_ca = qsum + midsum + assesment;


			-- insert into temp_result
			INSERT INTO temp_result
			 VALUES (
				reg_no,
				cname,
				FORMAT(final_ca,2)
			 );

			SET count = count + 1;
		END WHILE;
	END IF;
	SET maincount = maincount + 1;
END WHILE;


    -- final output
    SELECT * FROM temp_result ORDER BY registration_no;

    DROP TEMPORARY TABLE IF EXISTS temp_result;
END//

CREATE PROCEDURE individual_eligibility(IN reg_no CHAR(10))
BEGIN
    DECLARE total_courses INT DEFAULT 0;
    DECLARE counter INT DEFAULT 0;
    DECLARE cur_course VARCHAR(10);
    DECLARE sub_status VARCHAR(20);
    DECLARE stu_status VARCHAR(20);

    DECLARE q1, q2, q3, mt, mp, midsum, assessment, qmax, qsum, final_ca DECIMAL(5,2) DEFAULT 0;
    DECLARE total_sessions, attended_sessions, medical_sessions INT DEFAULT 0;
    DECLARE attendance_pct DECIMAL(5,2) DEFAULT 0;
    DECLARE eligibility_status VARCHAR(20) DEFAULT 'WH';

    DROP TEMPORARY TABLE IF EXISTS temp_result;
    CREATE TEMPORARY TABLE temp_result (
        Registration_No CHAR(10),
        Course_Code VARCHAR(10),
        Attendance_Percentage DECIMAL(5,2),
        Final_CA_Mark VARCHAR(10),
        Eligibility VARCHAR(20)
    );

    
	
	
    SELECT COUNT(DISTINCT course_code) INTO total_courses
    FROM marks
    WHERE registration_no = reg_no;

    SELECT status INTO stu_status
    FROM student
    WHERE registration_no = reg_no;

    WHILE counter < total_courses DO
        SELECT DISTINCT course_code INTO cur_course
        FROM marks
        WHERE registration_no = reg_no
        LIMIT 1 OFFSET counter;

        SET attendance_pct = 0, final_ca = 0, eligibility_status = 'WH';

        -- Proper Students
        IF (stu_status = 'proper') THEN
            SELECT COUNT(*) INTO total_sessions
            FROM attendance
            WHERE registration_no = reg_no AND course = cur_course;

            SELECT COUNT(*) INTO attended_sessions
            FROM attendance
            WHERE registration_no = reg_no AND course = cur_course AND status = 'Present';

            SELECT COUNT(m.medical_id) INTO medical_sessions
            FROM medical m
            JOIN attendance a ON a.attendance_id = m.attendance_id
            WHERE a.registration_no = reg_no AND a.course = cur_course;

            IF total_sessions > 0 THEN
                SET attendance_pct = ((attended_sessions + medical_sessions) / total_sessions) * 100;
            ELSE
                SET attendance_pct = 0;
            END IF;

            -- Calculate Final CA
            SELECT IFNULL(MAX(mark),0) INTO q1 FROM marks WHERE registration_no = reg_no AND course_code = cur_course AND status = 'Q1';
            SELECT IFNULL(MAX(mark),0) INTO q2 FROM marks WHERE registration_no = reg_no AND course_code = cur_course AND status = 'Q2';
            SELECT IFNULL(MAX(mark),0) INTO q3 FROM marks WHERE registration_no = reg_no AND course_code = cur_course AND status = 'Q3';
            SELECT IFNULL(MAX(mark),0) INTO mt FROM marks WHERE registration_no = reg_no AND course_code = cur_course AND status = 'MT';
            SELECT IFNULL(MAX(mark),0) INTO mp FROM marks WHERE registration_no = reg_no AND course_code = cur_course AND status = 'MP';
            SELECT IFNULL(MAX(mark),0)*0.05 INTO assessment FROM marks WHERE registration_no = reg_no AND course_code = cur_course AND status = 'A';

            SELECT course_type INTO sub_status FROM course WHERE course_code = cur_course;

            IF (q1 < q2 AND q1 < q3) THEN
                SET qmax = (q2 + q3)/2;
            ELSEIF (q2 < q3 AND q2 < q1) THEN
                SET qmax = (q3 + q1)/2;
            ELSE
                SET qmax = (q1 + q2)/2;
            END IF;

            SET qsum = qmax * 0.10;

            IF (sub_status = 'BOTH') THEN
                SET midsum = ((mt + mp)/2) * 0.20;
            ELSEIF (sub_status = 'T') THEN
                SET midsum = mt * 0.20;
            ELSEIF (sub_status = 'P') THEN
                SET midsum = mp * 0.20;
            ELSE
                SET midsum = 0;
            END IF;

            SET final_ca = qsum + midsum + assessment;

            IF (attendance_pct >= 80) THEN
                SET eligibility_status = 'Eligible';
            ELSE
                SET eligibility_status = 'Not Eligible';
            END IF;




        -- Repeat Students
        ELSEIF (stu_status = 'repeat') THEN
            SELECT IFNULL(MAX(mark),0) INTO q1 FROM marks WHERE registration_no = reg_no AND course_code = cur_course AND status = 'Q1';
            SELECT IFNULL(MAX(mark),0) INTO q2 FROM marks WHERE registration_no = reg_no AND course_code = cur_course AND status = 'Q2';
            SELECT IFNULL(MAX(mark),0) INTO q3 FROM marks WHERE registration_no = reg_no AND course_code = cur_course AND status = 'Q3';
            SELECT IFNULL(MAX(mark),0) INTO mt FROM marks WHERE registration_no = reg_no AND course_code = cur_course AND status = 'MT';
            SELECT IFNULL(MAX(mark),0) INTO mp FROM marks WHERE registration_no = reg_no AND course_code = cur_course AND status = 'MP';
            SELECT IFNULL(MAX(mark),0)*0.05 INTO assessment FROM marks WHERE registration_no = reg_no AND course_code = cur_course AND status = 'A';

            SELECT course_type INTO sub_status FROM course WHERE course_code = cur_course;

            IF (q1 < q2 AND q1 < q3) THEN
                SET qmax = (q2 + q3)/2;
            ELSEIF (q2 < q3 AND q2 < q1) THEN
                SET qmax = (q3 + q1)/2;
            ELSE
                SET qmax = (q1 + q2)/2;
            END IF;

            SET qsum = qmax * 0.10;

            IF (sub_status = 'BOTH') THEN
                SET midsum = ((mt + mp)/2) * 0.20;
            ELSEIF (sub_status = 'T') THEN
                SET midsum = mt * 0.20;
            ELSEIF (sub_status = 'P') THEN
                SET midsum = mp * 0.20;
            ELSE
                SET midsum = 0;
            END IF;

            SET final_ca = qsum + midsum + assessment;

            IF (final_ca >= 14) THEN
                SET eligibility_status = 'Eligible';
            ELSE
                SET eligibility_status = 'Not Eligible';
            END IF;




        -- Suspended Students
        ELSEIF (stu_status = 'suspended') THEN
            SET eligibility_status = 'WH';
        END IF;

        INSERT INTO temp_result
        VALUES (reg_no, cur_course, attendance_pct, IF(stu_status = 'suspended', 'WH', FORMAT(final_ca,2)), eligibility_status);

        SET counter = counter + 1;
    END WHILE;

    SELECT * FROM temp_result ORDER BY Course_Code;
END//

CREATE OR REPLACE VIEW indi_grade AS
SELECT
    registration_no,
    course_name,
	
	-- 3 types grading for repaet, proper, suspended
	
	
    CASE
        WHEN status = 'repeat' THEN
            CASE    
                WHEN total >= 80 THEN 'C'
                WHEN total >= 70 THEN 'C'
                WHEN total >= 65 THEN 'C-'
                WHEN total >= 60 THEN 'C'
                WHEN total >= 55 THEN 'C'
                WHEN total >= 50 THEN 'C'
                WHEN total >= 45 THEN 'C'
                WHEN total >= 40 THEN 'C'
                WHEN total >= 35 THEN 'C-'
                WHEN total >= 30 THEN 'D'
                ELSE 'E'
            END
        WHEN status = 'proper' THEN
            CASE    
                WHEN total >= 85 THEN 'A+'
                WHEN total >= 75 THEN 'A'
                WHEN total >= 70 THEN 'A-'
                WHEN total >= 65 THEN 'B+'
                WHEN total >= 60 THEN 'B'
                WHEN total >= 55 THEN 'B-'
                WHEN total >= 50 THEN 'C+'
                WHEN total >= 45 THEN 'C'
                WHEN total >= 40 THEN 'C-'
                WHEN total >= 35 THEN 'D'
                ELSE 'E'
            END
		WHEN status = 'suspended' THEN 'WH'
    END AS grade
	
FROM (

    SELECT
        m.registration_no,
        c.course_name,
        s.status,
		
		-- calculate the final marks
		
		
        CASE 
            WHEN c.course_type = 'BOTH' THEN 
                (
                    ((COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END), 0) +
                      COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END), 0) +
                      COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END), 0) -
                      LEAST(
                          COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END), 0),
                          COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END), 0),
                          COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END), 0))) / 2) * 0.10
                ) +
                (
                    ((COALESCE(MAX(CASE WHEN m.status='MT' THEN m.mark END), 0) +
                      COALESCE(MAX(CASE WHEN m.status='MP' THEN m.mark END), 0)) / 2) * 0.20
                ) +
                (
                    (COALESCE(MAX(CASE WHEN m.status='FT' THEN m.mark END), 0) +
                     (COALESCE(MAX(CASE WHEN m.status='FP' THEN m.mark END), 0)) / 2) * 0.65
                ) +
                (COALESCE(MAX(CASE WHEN m.status='A' THEN m.mark END), 0) * 0.05)
                
            WHEN c.course_type = 'T' THEN
                (
                    ((COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END), 0) +
                      COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END), 0) +
                      COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END), 0) -
                      LEAST(
                          COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END), 0),
                          COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END), 0),
                          COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END), 0))) / 2) * 0.10
				) +
                (COALESCE(MAX(CASE WHEN m.status='MT' THEN m.mark END), 0) * 0.20) +
                ((COALESCE(MAX(CASE WHEN m.status='FT' THEN m.mark END), 0) * 0.65)) +
                (COALESCE(MAX(CASE WHEN m.status='A' THEN m.mark END), 0) * 0.05)
                
            WHEN c.course_type = 'P' THEN
                (
                    ((COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END), 0) +
                      COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END), 0) +
                      COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END), 0) -
                      LEAST(
                          COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END), 0),
                          COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END), 0),
                          COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END), 0))) / 2) * 0.10
                ) +
                (COALESCE(MAX(CASE WHEN m.status='MP' THEN m.mark END), 0) * 0.20) +
                (COALESCE(MAX(CASE WHEN m.status='FP' THEN m.mark END), 0) * 0.65) +
                (COALESCE(MAX(CASE WHEN m.status='A' THEN m.mark END), 0) * 0.05)
        END AS total
    FROM marks m
    JOIN course c ON c.course_code = m.course_code
    JOIN student s ON s.registration_no = m.registration_no
    GROUP BY m.registration_no, c.course_name, s.status, c.course_type
) AS temp_table;














CREATE OR REPLACE VIEW sgpa_cgpa AS
SELECT
    registration_no,
    ROUND(SUM(grade_points * credits) / SUM(credits), 2) AS sgpa, -- calculate part ofsgpa
    ROUND(SUM(grade_points * credits) / SUM(credits), 2) AS cgpa  -- calculate part of cgpa
FROM (
    SELECT
        totals.registration_no,
        totals.credits,
		
		
		-- 2 types for repeat and proper student grade_point calculate and suspended going to be 0
		
		
        ROUND(
            CASE
                WHEN totals.status = 'repeat' THEN
                    CASE
                        WHEN totals.total >= 80 THEN 2.0
                        WHEN totals.total >= 70 THEN 2.0
                        WHEN totals.total >= 65 THEN 2.0
                        WHEN totals.total >= 60 THEN 2.0
                        WHEN totals.total >= 55 THEN 2.0
                        WHEN totals.total >= 50 THEN 2.0
                        WHEN totals.total >= 45 THEN 2.0
                        WHEN totals.total >= 40 THEN 2.0
                        WHEN totals.total >= 35 THEN 1.7
                        WHEN totals.total >= 30 THEN 1.3
                        ELSE 0
                    END
                WHEN totals.status = 'proper' THEN
                    CASE
                        WHEN totals.total >= 85 THEN 4.0
                        WHEN totals.total >= 75 THEN 4.0
                        WHEN totals.total >= 70 THEN 3.7
                        WHEN totals.total >= 65 THEN 3.3
                        WHEN totals.total >= 60 THEN 3.0
                        WHEN totals.total >= 55 THEN 2.7
                        WHEN totals.total >= 50 THEN 2.3
                        WHEN totals.total >= 45 THEN 2.0
                        WHEN totals.total >= 40 THEN 1.7
                        WHEN totals.total >= 35 THEN 1.3
                        ELSE 0
                    END
				WHEN totals.status = 'suspended' THEN 0
            END, 2
        ) AS grade_points
    FROM (
        SELECT
            m.registration_no,
            c.credits,
            s.status,
			
			
	-- calculate the final marks
			
			
            CASE
                WHEN c.course_type = 'BOTH' THEN
                    ((COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END),0) +
                      COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END),0) +
                      COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END),0)
                     - LEAST(
                         COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END),0),
                         COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END),0),
                         COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END),0))) / 2) * 0.10
							+ 
					((COALESCE(MAX(CASE WHEN m.status='MT' THEN m.mark END),0) +
                        COALESCE(MAX(CASE WHEN m.status='MP' THEN m.mark END),0)) / 2) * 0.20
                    + (COALESCE(MAX(CASE WHEN m.status='FT' THEN m.mark END),0) +
                        ((COALESCE(MAX(CASE WHEN m.status='FP' THEN m.mark END),0)) / 2) * 0.65)
                    + (COALESCE(MAX(CASE WHEN m.status='A' THEN m.mark END),0) * 0.05)
                WHEN c.course_type = 'T' THEN
                    ((COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END),0) +
                      COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END),0) +
                      COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END),0)
                     - LEAST(
                         COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END),0),
                         COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END),0),
                         COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END),0))) / 2) * 0.10
						+
					(COALESCE(MAX(CASE WHEN m.status='MT' THEN m.mark END),0) * 0.20)
                    + (COALESCE(MAX(CASE WHEN m.status='FT' THEN m.mark END),0) * 0.65)
                    + (COALESCE(MAX(CASE WHEN m.status='A' THEN m.mark END),0) * 0.05)
                WHEN c.course_type = 'P' THEN
                    ((COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END),0) +
                      COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END),0) +
                      COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END),0)
                     - LEAST(
                         COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END),0),
                         COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END),0),
                         COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END),0)
                     )) / 2) * 0.10
                    + (COALESCE(MAX(CASE WHEN m.status='MP' THEN m.mark END),0) * 0.20)
                    + (COALESCE(MAX(CASE WHEN m.status='FP' THEN m.mark END),0) * 0.65)
                    + (COALESCE(MAX(CASE WHEN m.status='A' THEN m.mark END),0) * 0.05)
                ELSE 0
            END AS total
        FROM marks m
        JOIN course c ON m.course_code = c.course_code
        JOIN student s ON m.registration_no = s.registration_no
        GROUP BY m.registration_no, c.course_name, c.credits, s.status, c.course_type
    ) AS totals
) AS grade_calculated
GROUP BY registration_no
ORDER BY registration_no;

		 a.course,

CREATE PROCEDURE ca_batch_summary(IN course VARCHAR(10))
BEGIN
	DECLARE total_students INT DEFAULT 0;
	DECLARE count INT  DEFAULT 0;
	DECLARE reg_no VARCHAR(20);
	DECLARE s_status VARCHAR(20);
	
	DECLARE q1, q2, q3, mid_marks, mt_mark,mp_mark,assess_marks, top_two_total, final_ca DECIMAL(5,2) DEFAULT 0;
	DECLARE ca VARCHAR(20);

	DROP TEMPORARY TABLE IF EXISTS temp_result;
    CREATE TEMPORARY TABLE temp_result (
        Registration_No VARCHAR(20),
        Course_Code VARCHAR(20),
        Q1 VARCHAR(20),
        Q2 VARCHAR(20),
        Q3 VARCHAR(20),
        Mid_Marks VARCHAR(20),
        Assessment_Marks VARCHAR(20),
        Final_CA_Mark VARCHAR(20),
        CA_Status VARCHAR(20),
		S_Status VARCHAR(20)
    );
	
	DROP TEMPORARY TABLE IF EXISTS temp_students;
	
	CREATE TEMPORARY TABLE temp_students AS
        SELECT DISTINCT s.registration_no
        FROM student s
        JOIN marks m ON s.registration_no = m.registration_no
        WHERE m.course_code = course;
		
		SELECT COUNT(*) INTO total_students FROM temp_students;
		
		WHILE count < total_students DO
		SELECT registration_no INTO reg_no
        FROM temp_students
        LIMIT 1 OFFSET count;
		
		SELECT status INTO s_status
		FROM student
		WHERE registration_no = reg_no 
		LIMIT 1;

		IF s_status = 'suspended' THEN
            INSERT INTO temp_result
            VALUES (reg_no, course, 'wh','wh','wh','wh','wh','wh','wh',s_status);
        ELSE


		SELECT IFNULL(MAX(mark),0) INTO Q1
		FROM marks
		WHERE registration_no = reg_no AND course_code = course AND status = 'Q1';
		
		SELECT IFNULL(MAX(mark),0) INTO Q2
		FROM marks
		WHERE registration_no = reg_no AND course_code = course AND status = 'Q2';

		SELECT IFNULL(MAX(mark),0) INTO Q3
		FROM marks
		WHERE registration_no = reg_no AND course_code = course AND status = 'Q3';

		SELECT IFNULL(MAX(mark),0) INTO mt_mark
		FROM marks
		WHERE registration_no = reg_no AND course_code = course AND status = 'MT';

		SELECT IFNULL(MAX(mark),0) INTO mp_mark		
		FROM marks
		WHERE registration_no = reg_no AND course_code = course AND status = 'MP';

		IF (mt_mark > 0 AND mp_mark > 0) THEN
		SET mid_marks = (mt_mark + mp_mark)/2;
		ELSE
		SET mid_marks = (mt_mark + mp_mark);
		END IF;

		SELECT IFNULL(SUM(mark),0)
        INTO assess_marks
        FROM marks
        WHERE registration_no = reg_no AND course_code = course
        AND status = 'A';
		
		if(Q1<Q2 && Q1<Q3) then
			set top_two_total=(q2+q3)/2;
	

		elseif(Q2<Q3 && Q2<Q1) then
			set top_two_total=(q3+q1)/2;
	
		else
			set top_two_total=(q1+q2)/2;
		end if;
		
		SET final_ca = (top_two_total * 0.10) + (mid_marks * 0.20) + (assess_marks * 0.05);

		IF final_ca > (35 * 0.40) THEN
            SET ca = 'CA_Pass';
        ELSE
            SET ca = 'CA_Fail';
        END IF;

		INSERT INTO temp_result
        VALUES (reg_no,course, q1, q2, q3, mid_marks, assess_marks, final_ca,ca,s_status);
	END IF;

	SET count = count + 1;
		
    END WHILE; 

	SELECT * FROM temp_result ORDER BY Registration_No;
	
	DROP TEMPORARY TABLE IF EXISTS temp_students;
    DROP TEMPORARY TABLE IF EXISTS temp_result;
END//

CREATE PROCEDURE ca_individual_summary(in id char(10),IN course VARCHAR(10))
BEGIN
 
	DECLARE q1, q2, q3, mid_marks, mt_mark,mp_mark,assess_marks, top_two_total, final_ca_status DECIMAL(5,2) DEFAULT 0;
	DECLARE s_status varchar(20);
	DECLARE ca VARCHAR(20);
	
	SELECT IFNULL(MAX(mark),0) INTO q1 
	FROM marks
	WHERE registration_no=id AND course_code=course AND status='Q1';

	SELECT IFNULL(MAX(mark),0) INTO q2 
	FROM marks
	WHERE registration_no=id AND course_code=course AND status='Q2';

	SELECT IFNULL(MAX(mark),0) INTO q3 
	FROM marks

		IF (mt_mark > 0 AND mp_mark > 0) THEN

CREATE PROCEDURE whole_eligibility_without_attendance(IN course VARCHAR(10))
BEGIN
    DECLARE total_students INT DEFAULT 0;
    DECLARE count INT DEFAULT 0;
    DECLARE reg_no VARCHAR(20);
    DECLARE s_status VARCHAR(20);

    DECLARE q1, q2, q3, mid_marks, mt_mark, mp_mark, assess_marks, top_two_total, final_ca DECIMAL(5,2) DEFAULT 0;
    DECLARE eli_status VARCHAR(20);

    
    DROP TEMPORARY TABLE IF EXISTS temp_result;
    CREATE TEMPORARY TABLE temp_result (
        Registration_No VARCHAR(20),
        Course_Code VARCHAR(20),
        Final_CA_Mark DECIMAL(5,2),
        Eligibility VARCHAR(20),
		S_status VARCHAR(20)
    );

    DROP TEMPORARY TABLE IF EXISTS temp_students;
    CREATE TEMPORARY TABLE temp_students AS
        SELECT DISTINCT s.registration_no
        FROM student s
        JOIN marks m ON s.registration_no = m.registration_no
        WHERE m.course_code = course;

    SELECT COUNT(*) INTO total_students FROM temp_students;

    
    WHILE count < total_students DO
        SELECT registration_no INTO reg_no
        FROM temp_students
        LIMIT 1 OFFSET count;

        
        SELECT status INTO s_status
        FROM student
        WHERE registration_no = reg_no
        LIMIT 1;

        
        IF s_status = 'repeat' THEN

            IF s_status = 'suspended' THEN
                INSERT INTO temp_result VALUES (reg_no, course, 'wh', 'wh',s_status);
            ELSE
                
                SELECT IFNULL(MAX(mark),0) INTO q1 FROM marks
                WHERE registration_no = reg_no AND course_code = course AND status = 'Q1';

                SELECT IFNULL(MAX(mark),0) INTO q2 FROM marks
                WHERE registration_no = reg_no AND course_code = course AND status = 'Q2';

                SELECT IFNULL(MAX(mark),0) INTO q3 FROM marks
                WHERE registration_no = reg_no AND course_code = course AND status = 'Q3';

                SELECT IFNULL(MAX(mark),0) INTO mt_mark FROM marks
                WHERE registration_no = reg_no AND course_code = course AND status = 'MT';

                SELECT IFNULL(MAX(mark),0) INTO mp_mark FROM marks
                WHERE registration_no = reg_no AND course_code = course AND status = 'MP';

                
                IF (mt_mark > 0 AND mp_mark > 0) THEN
                    SET mid_marks = (mt_mark + mp_mark)/2;
                ELSE
                    SET mid_marks = (mt_mark + mp_mark);
                END IF;

                
                SELECT IFNULL(SUM(mark),0)
                INTO assess_marks
                FROM marks
                WHERE registration_no = reg_no AND course_code = course
                AND status = 'A';

                
                IF (q1 < q2 AND q1 < q3) THEN
                    SET top_two_total = (q2 + q3)/2;
                ELSEIF (q2 < q3 AND q2 < q1) THEN
                    SET top_two_total = (q3 + q1)/2;
                ELSE
                    SET top_two_total = (q1 + q2)/2;
                END IF;

               
                SET final_ca = (top_two_total * 0.10) + (mid_marks * 0.20) + (assess_marks * 0.05);

                
                IF final_ca > (30 * 0.40) THEN
                    SET eli_status = 'Eligible';
                ELSE
                    SET eli_status = 'Not_Eligible';
                END IF;

                INSERT INTO temp_result VALUES (reg_no, course, final_ca, eli_status,s_status);
            END IF;

        END IF; 

        SET count = count + 1;
    END WHILE;

    
    
        SELECT * FROM temp_result ORDER BY Registration_No;
    

    DROP TEMPORARY TABLE IF EXISTS temp_students;
    DROP TEMPORARY TABLE IF EXISTS temp_result;
END//

CREATE OR REPLACE VIEW lecture_courses AS
SELECT 
    l.lecture_id,
    l.full_name AS lecturer_name,
    c.course_code,
    c.course_name,
    d.department_name
FROM lecture l
JOIN course c ON l.lecture_id = c.lecture
LEFT JOIN department d ON l.assing_department = d.department_id
ORDER BY d.department_name, l.full_name;



CREATE OR REPLACE VIEW total_marks AS
SELECT
    m.registration_no,
    c.course_code,
    c.course_name,
    CASE 
        WHEN s.status = 'suspended' THEN 'wh'
        ELSE ROUND(
            CASE 
                WHEN c.course_type = 'BOTH' THEN 
                    (
                        ((COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END), 0) +
                          COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END), 0) +
                          COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END), 0) -
                          LEAST(
                              COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END), 0),
                              COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END), 0),
                              COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END), 0)
                          )) / 2) * 0.10
                    ) +
                    (
                        ((COALESCE(MAX(CASE WHEN m.status='MT' THEN m.mark END), 0) +
                          COALESCE(MAX(CASE WHEN m.status='MP' THEN m.mark END), 0)) / 2) * 0.20
                    ) +
                    (
                        ((COALESCE(MAX(CASE WHEN m.status='FT' THEN m.mark END), 0) +
                          COALESCE(MAX(CASE WHEN m.status='FP' THEN m.mark END), 0)) / 2) * 0.65
                    ) +
                    (COALESCE(MAX(CASE WHEN m.status='A' THEN m.mark END), 0) * 0.05)

                WHEN c.course_type = 'T' THEN
                    (
                        ((COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END), 0) +
                          COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END), 0) +
                          COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END), 0) -
                          LEAST(
                              COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END), 0),
                              COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END), 0),
                              COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END), 0)
                          )) / 2) * 0.10
                    ) +
                    (COALESCE(MAX(CASE WHEN m.status='MT' THEN m.mark END), 0) * 0.20) +
                    (COALESCE(MAX(CASE WHEN m.status='FT' THEN m.mark END), 0) * 0.65) +
                    (COALESCE(MAX(CASE WHEN m.status='A' THEN m.mark END), 0) * 0.05)

                WHEN c.course_type = 'P' THEN
                    (
                        ((COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END), 0) +
                          COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END), 0) +
                          COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END), 0) -
                          LEAST(
                              COALESCE(MAX(CASE WHEN m.status='Q1' THEN m.mark END), 0),
                              COALESCE(MAX(CASE WHEN m.status='Q2' THEN m.mark END), 0),
                              COALESCE(MAX(CASE WHEN m.status='Q3' THEN m.mark END), 0)
                          )) / 2) * 0.10
                    ) +
                    (COALESCE(MAX(CASE WHEN m.status='MP' THEN m.mark END), 0) * 0.20) +
                    (COALESCE(MAX(CASE WHEN m.status='FP' THEN m.mark END), 0) * 0.65) +
                    (COALESCE(MAX(CASE WHEN m.status='A' THEN m.mark END), 0) * 0.05)
            END, 2
        )
    END AS total_marks,s.status as student_status
FROM marks m
JOIN course c ON c.course_code = m.course_code
JOIN student s ON s.registration_no = m.registration_no
GROUP BY m.registration_no, c.course_code;

inner join student s on s.registration_no=a.registration_no
inner join course c on c.course_code=a.course

where c.course_code=course
group by a.registration_no,s.full_name,c.course_name;

CREATE PROCEDURE UpdateMedicalApproval(
    IN p_registration_no VARCHAR(10),
    IN p_attendance_id INT
)
BEGIN
    UPDATE medical 
    SET approval_status = 'approved'
    WHERE attendance_id = p_attendance_id;
    
    UPDATE attendance 
    SET status = 'medical'
    WHERE attendance_id = p_attendance_id 
      AND registration_no = p_registration_no;
